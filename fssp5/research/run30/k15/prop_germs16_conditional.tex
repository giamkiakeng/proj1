% Every number below is certified or independently checked (research/LOG.md, section 30, and run30/k15/).
% The only open hypothesis is the unsatisfiability of X(g_1) and X(g_2), stated as such.

\begin{proposition}\label{prop:germs16}
Let $g_1$ and $g_2$ be the germs \texttt{germs/r16/r16\_02.txt} and \texttt{germs/r16/r16\_15.txt} of the
supplementary material (both of complexity $15$). For a germ $g$ let $X(g)$ be the formula that encodes, for the
table entries left free by $g$ (with the cells of $g$'s half-line shared between the lines),
\begin{enumerate}
  \item[(a)] the lines $2\le n\le 15$ synchronized at time $2n-2$: every cell is in the firing state at
        time $2n-2$, and no cell is in it earlier;
  \item[(b)] the early rows $\tau\le 12$ of every line $n\ge 16$, imposed through the representatives
        $16\le n\le 130$ given by Corollary~\ref{cor:reduce};
  \item[(c)] the right-end cones \textup{(G5)} of the lines $16\le n\le 20$.
\end{enumerate}
If $X(g_1)$ and $X(g_2)$ are unsatisfiable, then every five-state minimal-time solution $\delta$ satisfies
$c_{78}(\delta)\ge 16$.
\end{proposition}

\begin{proof}
Let $\delta$ be a five-state minimal-time solution with $c_{78}(\delta)\le 15$. By
Proposition~\ref{prop:germs}, $c_{78}(\delta)=15$. The depth-first search of the proof of
Proposition~\ref{prop:germs}, run with the bound $15$ instead of $14$ (no firing state in the germ, never $L$ at
the front, $\mathrm{PUMP}(n,n')$ for $4\le n<n'\le 40$), returns $1{,}635$ germs of complexity at most $15$:
$2$, $26$, $216$ and $1{,}391$ of complexity $12$, $13$, $14$ and $15$. An independent enumerator returns the
same $1{,}635$ germs. The germ of $\delta$ is one of the $1{,}391$ of complexity $15$, and each of these is
excluded by one of the following four arguments.
\begin{enumerate}
  \item[(i)] For $1{,}270$ germs, $\mathrm{MT}_5(\{2,\dots,10\};78,39)$ restricted to the germ is
        unsatisfiable. A single selector formula covers all of them ($18{,}306$ variables, $575{,}418$
        clauses). CaDiCaL refutes it with an LRAT proof that \texttt{lrat-check} verifies.
  \item[(ii)] For $117$ germs, the hypotheses of Lemma~\ref{lem:left} hold at an explicit witness:
        $102$ with $s=1$ and $25\le n\le 167$, and $15$ with $s=2$ and $388\le n\le 2565$, all with
        period $\ell\le 6$. A checker that shares no code with the search confirms each witness by
        simulating the half-line from the germ alone.
  \item[(iii)] For $2$ germs, $\mathrm{PUMP}(77,109)$, respectively $\mathrm{PUMP}(45,77)$, fails. This
        contradicts Lemma~\ref{lem:pumping}.
  \item[(iv)] The remaining $2$ germs are $g_1$ and $g_2$. Their half-line is the half-line of $\delta$.
        $\delta$ satisfies (a) by the definition of a minimal-time solution, and (c) by
        Lemma~\ref{lem:G}. By Lemma~\ref{lem:early} and
        Corollary~\ref{cor:reduce}, it satisfies (b). So the table of $\delta$ is a model of $X(g_1)$ or
        of $X(g_2)$, which contradicts the hypothesis.
\end{enumerate}
Since $1{,}391=1{,}270+117+2+2$, no germ of complexity $15$ occurs, a contradiction.
\end{proof}
